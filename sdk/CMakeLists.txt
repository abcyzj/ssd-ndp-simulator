cmake_minimum_required(VERSION 3.22)
project(ndpsim_capture VERSION 0.4.5 LANGUAGES CXX)
add_library(ndpsim_capture STATIC src/capture.cc src/sha256.cc)
add_library(ndpsim::capture ALIAS ndpsim_capture)
target_compile_features(ndpsim_capture PUBLIC cxx_std_17)
target_include_directories(ndpsim_capture PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(ndpsim_capture PRIVATE -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
target_link_libraries(ndpsim_capture PUBLIC Threads::Threads)
option(NDPSIM_CAPTURE_TESTING "Build SDK-only tests and metadata fixture" OFF)
if(NDPSIM_CAPTURE_TESTING)
  enable_testing()
  add_executable(capture_fixture tests/capture_fixture.cc)
  target_link_libraries(capture_fixture PRIVATE ndpsim::capture)
  add_test(NAME sdk_sha256 COMMAND capture_fixture --hash)
  add_test(NAME sdk_measurement_metadata COMMAND capture_fixture --sample)
endif()
include(GNUInstallDirs)
install(TARGETS ndpsim_capture EXPORT ndpsimCaptureTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT ndpsimCaptureTargets NAMESPACE ndpsim:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ndpsimCapture)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/ndpsim-capture)
set_target_properties(ndpsim_capture PROPERTIES EXPORT_NAME capture)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/ndpsimCaptureConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/ndpsimCaptureConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ndpsimCapture)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/ndpsimCaptureConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ndpsimCaptureConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/ndpsimCaptureConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ndpsimCapture)
