cmake_minimum_required(VERSION 3.22)
# This directory also builds independently with only the public SDK.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(ndpsim_block_sum LANGUAGES CXX)
  if(NOT NDPSIM_SDK_ROOT)
    message(FATAL_ERROR "Provide -DNDPSIM_SDK_ROOT=/path/to/sdk")
  endif()
  add_subdirectory("${NDPSIM_SDK_ROOT}" sdk)
endif()
add_executable(block_sum_native native.cc)
target_link_libraries(block_sum_native PRIVATE ndpsim::capture)
target_compile_features(block_sum_native PRIVATE cxx_std_17)
target_compile_options(block_sum_native PRIVATE -fno-fast-math -ffp-contract=off)
if(TARGET ndpsim_controller_timing)
  add_library(ndpsim_block_sum MODULE sst.cc)
  target_include_directories(ndpsim_block_sum PRIVATE "${PROJECT_BINARY_DIR}/generated" "${PROJECT_SOURCE_DIR}/third_party/sst-core/external")
  target_link_libraries(ndpsim_block_sum PRIVATE ndpsim_dag ndpsim_controller_timing ndpsim_backend SST::SSTCore)
  target_compile_features(ndpsim_block_sum PRIVATE cxx_std_17)
  if(APPLE)
    target_link_options(ndpsim_block_sum PRIVATE "-undefined" "dynamic_lookup")
  endif()
  if(BUILD_TESTING AND NDPSIM_SANITIZE)
    add_executable(block_sum_probe probe.cc)
    target_include_directories(block_sum_probe PRIVATE "${PROJECT_BINARY_DIR}/generated" "${PROJECT_SOURCE_DIR}/third_party/sst-core/external")
    target_link_libraries(block_sum_probe PRIVATE ndpsim_dag ndpsim_controller_timing ndpsim_backend)
    add_test(NAME extension_v045 COMMAND "${CMAKE_COMMAND}" -E env NDPSIM_EXTENSION_NATIVE=1 "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_extension.py" "${PROJECT_BINARY_DIR}")
    set_tests_properties(extension_v045 PROPERTIES TIMEOUT 180)
  endif()
  if(BUILD_TESTING AND NOT NDPSIM_SANITIZE)
    add_test(NAME extension_v045 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_extension.py" "${PROJECT_BINARY_DIR}")
    set_tests_properties(extension_v045 PROPERTIES TIMEOUT 180)
  endif()
endif()
