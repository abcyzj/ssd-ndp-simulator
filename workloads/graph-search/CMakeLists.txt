cmake_minimum_required(VERSION 3.22)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_SYSROOT AND NOT DEFINED ENV{SDKROOT})
    execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE _sdk
        OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    set(CMAKE_OSX_SYSROOT "${_sdk}" CACHE PATH "macOS SDK")
endif()
project(ndpsim_graph_search VERSION 0.4.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
set(NDPSIM_SDK_ROOT "" CACHE PATH "Explicit capture SDK source directory; empty uses find_package")
if(NDPSIM_SDK_ROOT)
    get_filename_component(NDPSIM_SDK_ROOT "${NDPSIM_SDK_ROOT}" ABSOLUTE)
    if(NOT EXISTS "${NDPSIM_SDK_ROOT}/include/ndpsim/capture.h")
        message(FATAL_ERROR "NDPSIM_SDK_ROOT must contain the capture SDK source")
    endif()
    add_subdirectory("${NDPSIM_SDK_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/capture-sdk")
else()
    find_package(ndpsimCapture 0.4.5 CONFIG REQUIRED)
endif()
add_library(ndpsim_graph_values STATIC src/content/value.cc src/content/hash.cc)
target_include_directories(ndpsim_graph_values PUBLIC src)
target_link_libraries(ndpsim_graph_values PUBLIC ndpsim::capture)
add_library(ndpsim_graph_operators STATIC src/operators/operators.cc)
target_link_libraries(ndpsim_graph_operators PUBLIC ndpsim_graph_values)
add_library(ndpsim_graph_algorithm STATIC src/search/core.cc)
target_link_libraries(ndpsim_graph_algorithm PUBLIC ndpsim_graph_operators)
add_executable(ndpsim_native src/search/native.cc)
find_package(Threads REQUIRED)
target_link_libraries(ndpsim_native PRIVATE ndpsim_graph_algorithm Threads::Threads)
foreach(_target ndpsim_graph_values ndpsim_graph_operators ndpsim_graph_algorithm ndpsim_native)
    target_compile_options(${_target} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endforeach()
# Identity of actual source inputs, SDK and selected compiler/flags. Reconfigure on edits.
file(GLOB_RECURSE _identity_files CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" src/*)
list(PREPEND _identity_files CMakeLists.txt cmake/build-identity.cmake.in)
set(_identity "")
foreach(_source IN LISTS _identity_files)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${_source}")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${_source}" _sha)
    string(APPEND _identity "${_source}=${_sha}\n")
endforeach()
# Bind the actual linked SDK archive, also for installed package consumers.
string(SHA256 _source_sha "${_identity}")
configure_file(cmake/build-identity.cmake.in build-identity.cmake @ONLY)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-identity.h"
    COMMAND "${CMAKE_COMMAND}" "-DSDK_ARCHIVE=$<TARGET_FILE:ndpsim::capture>"
        "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/native-identity.h"
        -P "${CMAKE_CURRENT_BINARY_DIR}/build-identity.cmake"
    DEPENDS ndpsim::capture "$<TARGET_FILE:ndpsim::capture>" "${CMAKE_CURRENT_BINARY_DIR}/build-identity.cmake"
    VERBATIM)
target_sources(ndpsim_native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/native-identity.h")
target_compile_options(ndpsim_native PRIVATE -include "${CMAKE_CURRENT_BINARY_DIR}/native-identity.h")
set(_flags "${CMAKE_CXX_FLAGS} -std=c++17 -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off")
foreach(_config Debug Release MinSizeRel RelWithDebInfo)
    string(TOUPPER "${_config}" _upper)
    string(APPEND _flags " $<$<CONFIG:${_config}>:${CMAKE_CXX_FLAGS_${_upper}}>")
endforeach()
target_compile_definitions(ndpsim_native PRIVATE NDPSIM_NATIVE_FLAGS="${_flags}")
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    add_executable(search_core_probe tests/search_core_probe.cc)
    target_link_libraries(search_core_probe PRIVATE ndpsim_graph_algorithm)
    target_compile_options(search_core_probe PRIVATE -fno-fast-math -ffp-contract=off)
    add_test(NAME graph_numeric_search_work COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_search_core.py" --probe "$<TARGET_FILE:search_core_probe>")
    add_test(NAME graph_native_batch COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_search.py" "$<TARGET_FILE:ndpsim_native>")
endif()
if(BUILD_TESTING)
    add_test(NAME graph_raw_observation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_observation.py" "$<TARGET_FILE:ndpsim_native>")
endif()
